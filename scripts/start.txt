# Стартовый скрипт эмулятора (Вариант №6).
# Поддерживаются комментарии: строки, начинающиеся с #.

# Базовые команды
ls
ls -la

# Навигация
cd /tmp
cd ..
cd /

# Ошибочная строка — скрипт остановится здесь
